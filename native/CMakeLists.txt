cmake_minimum_required(VERSION 3.16)
project(rawtojxl-rawspeed-cli CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(MSVC)
  add_compile_options(/Zc:preprocessor /EHsc)
endif()

add_subdirectory(rawspeed)

add_executable(rawspeed-cli rawspeed-cli/main.cpp)
target_link_libraries(rawspeed-cli PRIVATE rawspeed)
target_include_directories(rawspeed-cli PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/rawspeed/src)
if(MSVC)
  target_compile_options(rawspeed-cli PRIVATE "/FI${CMAKE_CURRENT_SOURCE_DIR}/rawspeed-cli/msvc_compat.h")
  get_target_property(_rawspeed_targets rawspeed LINK_LIBRARIES)
  foreach(_t IN LISTS _rawspeed_targets)
    if(TARGET ${_t})
      get_target_property(_ttype ${_t} TYPE)
      if(_ttype MATCHES "^(STATIC_LIBRARY|SHARED_LIBRARY|MODULE_LIBRARY|EXECUTABLE|OBJECT_LIBRARY)$")
        target_compile_options(${_t} PRIVATE "/FI${CMAKE_CURRENT_SOURCE_DIR}/rawspeed-cli/msvc_compat.h")
      endif()
    endif()
  endforeach()
endif()
